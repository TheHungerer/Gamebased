<script>
  const users = {};

  function show(id) {
    document.querySelectorAll('.page').forEach(p => p.classList.remove('active'));
    const pg = document.getElementById(id);
    if (pg) pg.classList.add('active');
  }

  function err(id, msg) { document.getElementById(id).textContent = msg; }

  function doLogin() {
    const u = document.getElementById('l-user').value.trim();
    const p = document.getElementById('l-pass').value;
    err('l-err','');
    if (!u || !p) { err('l-err','Please fill in all fields.'); return; }
    if (!users[u]) { err('l-err','No account found. Please sign up first.'); return; }
    if (users[u].password !== p) { err('l-err','Incorrect password.'); return; }
    // Redirect to main.html after successful login
    window.location.href = 'main.html';
  }

  function doSignup() {
    const email = document.getElementById('s-email').value.trim();
    const u = document.getElementById('s-user').value.trim();
    const p = document.getElementById('s-pass').value;
    const c = document.getElementById('s-conf').value;
    err('s-err','');
    if (!email || !u || !p || !c) { err('s-err','Please fill in all fields.'); return; }
    if (!/\S+@\S+\.\S+/.test(email)) { err('s-err','Enter a valid email address.'); return; }
    if (p.length < 6) { err('s-err','Password must be at least 6 characters.'); return; }
    if (p !== c) { err('s-err','Passwords do not match.'); return; }
    if (users[u]) { err('s-err','Username already taken.'); return; }
    users[u] = { email, password: p };
    document.getElementById('new-username').textContent = u;
    document.getElementById('l-user').value = u;
    document.getElementById('l-pass').value = '';
    show('pg-success');
  }

  document.addEventListener('keydown', e => {
    if (e.key !== 'Enter') return;
    const active = document.querySelector('.page.active');
    if (!active) return;
    if (active.id === 'pg-login') doLogin();
    else if (active.id === 'pg-signup') doSignup();
    else if (active.id === 'pg-success') show('pg-login');
  });
</script>
