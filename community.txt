<script>
  const box       = document.getElementById('chat-box');
  const msgInput  = document.getElementById('message-input');
  const userInput = document.getElementById('username');

  const avatarEmojis = ['🧑‍💻','👩‍💻','🧙','🤖','👾','🦾','🐱‍💻','⚡','🔮','🧪'];
  const avatarColors = [
    'linear-gradient(135deg,#003a5a,#0099cc)',
    'linear-gradient(135deg,#1a004a,#7b2ff7)',
    'linear-gradient(135deg,#004a1a,#00c853)',
    'linear-gradient(135deg,#4a0000,#c62828)',
    'linear-gradient(135deg,#004a3a,#00bfa5)',
    'linear-gradient(135deg,#3a2a00,#f9a825)',
  ];
  const langColors = { java:'#f89820', js:'#f7df1e', html:'#e34c26', css:'#4a90e2', general:'#00d4ff' };
  const langBg     = { java:'rgba(248,152,32,0.12)', js:'rgba(247,223,30,0.1)', html:'rgba(227,76,38,0.12)', css:'rgba(74,144,226,0.12)', general:'rgba(0,212,255,0.1)' };
  const langBorder = { java:'rgba(248,152,32,0.28)', js:'rgba(247,223,30,0.25)', html:'rgba(227,76,38,0.28)', css:'rgba(74,144,226,0.28)', general:'rgba(0,212,255,0.25)' };
  const badges     = ['NEWBIE','STUDENT','STUDENT','HELPER','HELPER','PRO'];
  const badgeClass = { NEWBIE:'badge-newbie', STUDENT:'badge-student', HELPER:'badge-helper', PRO:'badge-pro' };
  const nameColors = ['#7eb8f7','#74c69d','#e8c97a','#c39bd3','#f0a500','#7fd8be','#ff8a70','#00d4ff'];

  let selectedLang = 'general';
  let filterActive = 'all';
  let msgCount = 6;

  function hash(str) { return [...str].reduce((a,c)=>a+c.charCodeAt(0),0); }
  function getAv(name)   { const h=hash(name); return { emoji:avatarEmojis[h%avatarEmojis.length], color:avatarColors[h%avatarColors.length] }; }
  function getBadge(name){ const b=badges[hash(name)%badges.length]; return {label:b, cls:badgeClass[b]}; }
  function getNC(name)   { return nameColors[hash(name)%nameColors.length]; }
  function getTime()     { const n=new Date(); return n.getHours().toString().padStart(2,'0')+':'+n.getMinutes().toString().padStart(2,'0'); }
  function esc(s)        { return s.replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;'); }

  function updatePreview() {
    const name = userInput.value.trim()||'?';
    const av = getAv(name);
    const el = document.getElementById('preview-avatar');
    el.textContent = av.emoji; el.style.background = av.color;
    const sav = document.getElementById('my-sidebar-av');
    sav.textContent = av.emoji; sav.style.background = av.color;
    document.getElementById('my-sidebar-name').textContent = name;
  }

  function pickLang(el) {
    document.querySelectorAll('.lp').forEach(e=>e.classList.remove('selected'));
    el.classList.add('selected');
    selectedLang = el.dataset.lang;
  }

  function filterLang(lang, el) {
    filterActive = lang;
    document.querySelectorAll('.chip').forEach(e=>e.classList.remove('active'));
    el.classList.add('active');
    document.querySelectorAll('.msg[data-lang]').forEach(m=>{
      m.style.display = (lang==='all'||m.dataset.lang===lang)?'':'none';
    });
  }

  function addEmoji(e) { msgInput.value+=e; msgInput.focus(); }

  function sendMessage() {
    const user = userInput.value.trim();
    const text = msgInput.value.trim();
    if (!user) { userInput.focus(); userInput.parentElement.style.borderColor='rgba(227,76,38,0.5)'; setTimeout(()=>userInput.parentElement.style.borderColor='',1200); return; }
    if (!text) { msgInput.focus(); return; }

    const av    = getAv(user);
    const badge = getBadge(user);
    const nc    = getNC(user);
    const lc    = langColors[selectedLang]||'#00d4ff';
    const lb    = langBg[selectedLang]||'rgba(0,212,255,0.1)';
    const lbrd  = langBorder[selectedLang]||'rgba(0,212,255,0.25)';
    const lname = selectedLang.charAt(0).toUpperCase()+selectedLang.slice(1);
    const time  = getTime();

    const div = document.createElement('div');
    div.className = 'msg own';
    div.dataset.lang = selectedLang;
    if (filterActive!=='all' && filterActive!==selectedLang) div.style.display='none';

    div.innerHTML = `
      <div class="avatar" style="background:${av.color}">${av.emoji}</div>
      <div class="msg-body">
        <div class="msg-meta">
          <span class="msg-name" style="color:${nc}">${esc(user)}</span>
          <span class="rank-badge ${badge.cls}">${badge.label}</span>
          <span class="lang-tag" style="background:${lb};color:${lc};border:1px solid ${lbrd}">${lname}</span>
          <span class="msg-time">${time}</span>
        </div>
        <div class="msg-bubble">${esc(text)}</div>
      </div>
      <div class="reactions"><button class="r-btn">👍</button><button class="r-btn">🔥</button><button class="r-btn">❤️</button><button class="r-btn">🎯</button></div>
    `;
    box.appendChild(div);
    box.scrollTop = box.scrollHeight;
    msgInput.value = '';
    msgInput.style.height = 'auto';
    msgInput.focus();

    msgCount++;
    document.getElementById('msg-count').textContent = msgCount;
  }

  msgInput.addEventListener('input', function(){
    this.style.height='auto';
    this.style.height=Math.min(this.scrollHeight,100)+'px';
  });
  msgInput.addEventListener('keydown', e=>{
    if (e.key==='Enter' && !e.shiftKey) { e.preventDefault(); sendMessage(); }
  });

  box.scrollTop = box.scrollHeight;
</script>
