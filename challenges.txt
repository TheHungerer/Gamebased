/* ─── LEVEL META ─── */
const LEVEL_META = [
  { title: 'HTML: Basics',          lang: 'html' },
  { title: 'CSS: Style a Button',   lang: 'css'  },
  { title: 'JS: Variables & Types', lang: 'js'   },
  { title: 'HTML: Links & Images',  lang: 'html' },
  { title: 'CSS: Flexbox Layout',   lang: 'css'  },

  { title: 'CSS: Grid Layout',      lang: 'css'  },
  { title: 'JS: DOM & Events',      lang: 'js'   },
];
const TOTAL_LEVELS = 8;

/* ─── STATE ─── */
let unlockedUpTo = 1;
let completedLevels = new Set();
let currentLevel = 1;
let currentQ = 0;
let selectedOpt = null;
let answered = false;
let correctCount = 0;
let totalAnswered = 0;
let xp = 0;
let streak = 0;

/* ─── RENDER MAP ─── */
function renderMap() {
  const track = document.getElementById('map-track');
  track.innerHTML = '';
  const done = completedLevels.size;
  document.getElementById('global-prog-label').textContent = `${done} / ${TOTAL_LEVELS} Levels Complete`;
  document.getElementById('global-prog-fill').style.width = ((done / TOTAL_LEVELS) * 100) + '%';

  for (let lvl = 1; lvl <= TOTAL_LEVELS; lvl++) {
    const meta = LEVEL_META[lvl - 1];
    const isDone    = completedLevels.has(lvl);
    const isCurrent = !isDone && lvl === unlockedUpTo;
    const isLocked  = lvl > unlockedUpTo;

    if (lvl > 1) {
      const conn = document.createElement('div');
      conn.className = 'connector' + (completedLevels.has(lvl - 1) ? ' active' : '');
      track.appendChild(conn);
    }

    const node = document.createElement('div');
    node.className = 'level-node';
    node.style.animationDelay = ((lvl - 1) * 0.05) + 's';

    if (!isLocked) {
      node.onclick = () => openChallenge(lvl);
    }

    if (isCurrent) {
      const arrow = document.createElement('div');
      arrow.className = 'play-arrow';
      node.appendChild(arrow);
    }

    const circle = document.createElement('div');
    if (isDone)         circle.className = 'node-circle done';
    else if (isCurrent) circle.className = 'node-circle current';
    else                circle.className = 'node-circle locked';

    if (isDone)         circle.textContent = '✓';
    else if (isCurrent) circle.textContent = lvl;
    else                circle.textContent = '🔒';
    node.appendChild(circle);

    const stars = document.createElement('div');
    stars.className = 'node-stars';
    for (let s = 0; s < 3; s++) {
      const star = document.createElement('span');
      star.className = 'star ' + (isDone ? 'lit' : 'dim');
      star.textContent = '★';
      stars.appendChild(star);
    }
    node.appendChild(stars);

    const label = document.createElement('div');
    label.className = 'node-label ' + (isDone ? 'done-label' : isCurrent ? 'current-label' : 'locked-label');
    label.textContent = meta.title;
    node.appendChild(label);

    track.appendChild(node);
  }
}

/* ─── VIEWS ─── */
function showMap() {
  renderMap();
  document.getElementById('map-view').style.display = 'flex';
  document.getElementById('challenge-view').style.display = 'none';
}

function openChallenge(lvl) {
  if (lvl > unlockedUpTo) return;
  currentLevel = lvl;
  currentQ = 0;
  selectedOpt = null;
  answered = false;

  document.getElementById('map-view').style.display = 'none';
  document.getElementById('challenge-view').style.display = 'flex';

  const data = CHALLENGES[lvl];
  document.getElementById('ch-level-label').textContent = 'Level ' + lvl + ':';
  document.getElementById('ch-level-title').textContent = data ? data.title : LEVEL_META[lvl-1].title;

  resetScores();
  loadQuestion();
}

/* ─── LOAD QUESTION ─── */
function loadQuestion() {
  const data = CHALLENGES[currentLevel];
  if (!data) return;

  const gi = document.getElementById('guide-items');
  gi.innerHTML = '';
  data.guide.forEach((g, i) => {
    gi.innerHTML += `<div class="guide-item" style="animation-delay:${i*0.07}s">
      <div class="guide-num">${i+1}</div>
      <div class="guide-text">${g.text}</div>
    </div>`;
  });

  document.getElementById('hint-text').textContent = data.hint;

  const total = data.questions.length;
  const pct = (currentQ / total) * 100;
  document.getElementById('level-prog-fill').style.width = pct + '%';
  document.getElementById('level-prog-label').textContent = `Q ${currentQ + 1} / ${total}`;

  const q = data.questions[currentQ];
  document.getElementById('problem-q').textContent = q.q;

  const ow = document.getElementById('options-wrap');
  ow.innerHTML = '';
  const letters = ['A', 'B', 'C', 'D'];
  q.opts.forEach((opt, i) => {
    const div = document.createElement('div');
    div.className = 'option';
    div.id = 'opt-' + i;
    div.onclick = () => selectOption(i);
    div.innerHTML = `<div class="opt-letter">${letters[i]}</div>`;
    const span = document.createElement('div');
    span.className = 'opt-text';
    span.textContent = opt;
    div.appendChild(span);
    ow.appendChild(div);
  });

  document.getElementById('feedback').className = 'feedback';
  document.getElementById('feedback').textContent = '';
  document.getElementById('btn-submit').disabled = true;
  document.getElementById('btn-submit').style.display = 'flex';
  document.getElementById('btn-next').classList.remove('show');
  document.getElementById('output-terminal').innerHTML =
    '<span class="out-comment">// Output will appear here after submission</span>';
  answered = false;
  selectedOpt = null;
}

function selectOption(i) {
  if (answered) return;
  document.querySelectorAll('.option').forEach(o => o.classList.remove('selected'));
  document.getElementById('opt-' + i).classList.add('selected');
  selectedOpt = i;
  document.getElementById('btn-submit').disabled = false;
}

function submitAnswer() {
  if (selectedOpt === null || answered) return;
  answered = true;
  totalAnswered++;

  const data = CHALLENGES[currentLevel];
  const q = data.questions[currentQ];
  const correct = q.correct;
  const isCorrect = selectedOpt === correct;

  document.querySelectorAll('.option').forEach(o => o.classList.add('disabled'));
  document.getElementById('opt-' + correct).classList.add('correct');
  if (!isCorrect) document.getElementById('opt-' + selectedOpt).classList.add('wrong');

  const fb = document.getElementById('feedback');
  if (isCorrect) {
    fb.textContent = '✓ Correct! ' + (currentQ < data.questions.length - 1 ? 'Keep going!' : 'Level complete! 🎉');
    fb.className = 'feedback correct';
    correctCount++;
    streak++;
    xp += 50;
  } else {
    fb.textContent = `✗ Not quite! The correct answer was: "${q.opts[correct]}"`;
    fb.className = 'feedback wrong';
    streak = 0;
  }

  const term = document.getElementById('output-terminal');
  if (isCorrect) {
    term.innerHTML = `<span class="out-comment">// Running answer check...</span>\n<span class="out-success">✓ PASS — Answer validated</span>\n<span class="out-info">// XP +50 awarded</span>\n<span class="out-info">// Streak: ${streak} 🔥</span>`;
  } else {
    term.innerHTML = `<span class="out-comment">// Running answer check...</span>\n<span class="out-error">✗ FAIL — Incorrect answer</span>\n<span class="out-comment">// Tip: ${data.hint}</span>`;
  }

  updateStats();

  const isLast = currentQ === data.questions.length - 1;
  document.getElementById('btn-submit').style.display = 'none';
  const nextBtn = document.getElementById('btn-next');
  nextBtn.textContent = isLast ? '🏆 Finish Level' : 'Next →';
  nextBtn.classList.add('show');
}

function nextQuestion() {
  const data = CHALLENGES[currentLevel];
  currentQ++;
  document.getElementById('btn-submit').style.display = 'flex';
  document.getElementById('btn-submit').disabled = true;
  if (currentQ >= data.questions.length) {
    finishLevel();
    return;
  }
  loadQuestion();
}

function skipQuestion() {
  if (answered) return;
  totalAnswered++;
  streak = 0;
  document.getElementById('feedback').textContent = '⏭ Skipped. Keep practising!';
  document.getElementById('feedback').className = 'feedback wrong';
  answered = true;
  currentQ++;
  const data = CHALLENGES[currentLevel];
  if (currentQ >= data.questions.length) { finishLevel(); return; }
  setTimeout(loadQuestion, 800);
}

function finishLevel() {
  const acc = totalAnswered ? Math.round((correctCount / totalAnswered) * 100) : 0;
  document.getElementById('output-terminal').innerHTML =
    `<span class="out-success">🏆 LEVEL COMPLETE!</span>\n<span class="out-info">// Score: ${correctCount}/${CHALLENGES[currentLevel].questions.length}</span>\n<span class="out-info">// Accuracy: ${acc}%</span>\n<span class="out-info">// Total XP: +${correctCount * 50}</span>`;

  document.getElementById('achievement').classList.add('show');
  document.getElementById('btn-next').classList.remove('show');
  document.getElementById('btn-submit').style.display = 'none';
  updateStats();

  completedLevels.add(currentLevel);
  if (currentLevel < TOTAL_LEVELS) {
    unlockedUpTo = currentLevel + 1;
  }

  setTimeout(() => {
    showMap();
    const t = document.getElementById('unlock-toast');
    if (currentLevel < TOTAL_LEVELS) {
      document.getElementById('toast-text').textContent = `Level ${currentLevel} complete! Level ${currentLevel + 1} unlocked 🔓`;
    } else {
      document.getElementById('toast-text').textContent = `All levels complete! You're a CodeQuest master! 🏆`;
    }
    t.style.display = 'flex';
    setTimeout(() => { t.style.display = 'none'; }, 5000);
  }, 2800);
}

function updateStats() {
  const acc = totalAnswered ? Math.round((correctCount / totalAnswered) * 100) : 100;
  document.getElementById('score-correct').textContent = correctCount;
  document.getElementById('score-xp').textContent = '+' + (correctCount * 50);
  document.getElementById('score-streak').textContent = '🔥' + streak;
  document.getElementById('score-acc').textContent = acc + '%';
  const maxXP = (CHALLENGES[currentLevel]?.questions?.length || 4) * 50;
  const xpPct = maxXP ? Math.min((xp / maxXP) * 100, 100) : 0;
  document.getElementById('xp-fill').style.width = xpPct + '%';
  document.getElementById('xp-label').textContent = xp + ' / ' + maxXP + ' XP';
}

function resetScores() {
  correctCount = 0;
  totalAnswered = 0;
  streak = 0;
  xp = 0;
  document.getElementById('achievement').classList.remove('show');
  updateStats();
}

showMap();
