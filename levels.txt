    const tiers = ["Beginner", "Intermediate", "Advanced"];
    let currentFilter = "all";

    function getCardClass(lv) {
      let cls = "lv-card " + lv.status;
      if (lv.boss) cls += " boss";
      return cls;
    }

    function getBadge(lv) {
      if (lv.status === "completed") return `<span class="card-badge badge-done">✓ DONE</span>`;
      if (lv.status === "active")    return `<span class="card-badge badge-active">IN PROGRESS</span>`;
      if (lv.boss)                   return `<span class="card-badge badge-boss">BOSS</span>`;
      if (lv.status === "locked")    return `<span class="card-badge badge-locked">🔒</span>`;
      return `<span class="card-badge badge-play">PLAY</span>`;
    }

    function renderLevels() {
      const wrap = document.getElementById("pathWrap");
      wrap.innerHTML = "";

      tiers.forEach(tier => {
        const tierLevels = levels.filter(lv => {
          if (lv.tier !== tier) return false;
          if (currentFilter === "all")       return true;
          if (currentFilter === "boss")      return lv.boss;
          if (currentFilter === "completed") return lv.status === "completed";
          if (currentFilter === "available") return lv.status === "available" || lv.status === "active";
          if (currentFilter === "locked")    return lv.status === "locked";
          return true;
        });
        if (!tierLevels.length) return;

        const block = document.createElement("div");
        block.className = "tier-block";
        block.innerHTML = `<div class="tier-label">${tier.toUpperCase()}</div>`;

        const grid = document.createElement("div");
        grid.className = "levels-grid";

        tierLevels.forEach(lv => {
          const card = document.createElement("div");
          card.className = getCardClass(lv);
          if (lv.status !== "locked") card.onclick = () => openModal(lv);
          card.innerHTML = `
            <div class="card-top">
              <span class="card-num">LV.${String(lv.id).padStart(2, "0")}</span>
              ${getBadge(lv)}
            </div>
            <span class="card-icon">${lv.icon}</span>
            <div class="card-name">${lv.name}</div>
            <div class="card-xp">+${lv.xp} XP · ${lv.mins}min</div>
            ${lv.progress > 0 ? `<div class="mini-prog"><div class="mini-fill" style="width:${lv.progress}%"></div></div>` : ""}
          `;
          grid.appendChild(card);
        });

        block.appendChild(grid);
        wrap.appendChild(block);
      });
    }

    function setFilter(f, btn) {
      currentFilter = f;
      document.querySelectorAll(".filter-btn").forEach(b => b.classList.remove("active"));
      btn.classList.add("active");
      renderLevels();
    }

        ${isActive ? `
          <div class="modal-prog-wrap">
            <div class="modal-prog-meta"><span>Progress</span><span>${lv.progress}%</span></div>
            <div class="modal-prog-track"><div class="modal-prog-fill" style="width:${lv.progress}%"></div></div>
          </div>` : ""}
        <button class="${btnClass}" onclick="alert('Navigating to level ${lv.id}...')">${btnLabel}</button>
      `;
      document.getElementById("modalBackdrop").classList.add("open");
    }

    function closeModal(e) {
      if (e.target === document.getElementById("modalBackdrop")) closeModalDirect();
    }

    function closeModalDirect() {
      document.getElementById("modalBackdrop").classList.remove("open");
    }

    renderLevels();
  </script>
